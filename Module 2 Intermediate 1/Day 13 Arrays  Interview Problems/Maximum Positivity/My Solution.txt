Time : O(N)
public class Solution 
{
    public int[] solve(int[] arr)
    {
        int idx = -1;
        int N = arr.length;
        int len = 0;
        int maxLen = 0;
        int[] ans;

        for(int i = 0; i < N; i++)
        {
            if(arr[i] >= 0)
            {
                len++;

                if(maxLen < len)
                {
                    maxLen = len;
                    idx = i + 1 - maxLen;
                }
            }else
            {
                len = 0;
            }
        }

        ans = new int[maxLen];

        for(int i = 0; i < maxLen; i++, idx++)
        {
            ans[i] = arr[idx];
        }

        return ans;
    }
}
