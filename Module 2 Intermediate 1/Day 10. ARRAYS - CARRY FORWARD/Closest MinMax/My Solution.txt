================Solution 1================
Time: O(N^2)
Space: O(1)
public class Solution 
{
    public int solve(int[] arr)
    {
        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        int N = arr.length;
        int leftLen = N;
        int rightLen = N;
        int len = N;

        for(int e : arr)
        {
            if(e > max)
            {
                max = e;
            }else if(e < min)
            {
                min = e;
            }
        }

        if(min == max)
        {
            return 1;
        }

        for(int i = 0; i < N; i++)
        {
            if(arr[i] == max)
            {
                for(int l = i - 1; l >= 0; l--)
                {
                    if(arr[l] == min)
                    {
                        leftLen = Math.min((i - l + 1), len);
                        break;
                    }
                }

                for(int r = i + 1; r < N; r++)
                {
                    if(arr[r] == min)
                    {
                        rightLen = Math.min((r - i + 1), len);
                        break;
                    }
                }

                len = Math.min(len, Math.min(leftLen, rightLen));
            }
        }

        return len;
    }
}
================Solution 2================
Time: O(1)
Space: O(1)
public class Solution 
{
    public int solve(int[] arr)
    {
        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        int N = arr.length;
        int minIdx = -1;
        int maxIdx = -1;
        int len = N;

        for(int e : arr)
        {
            if(e > max)
            {
                max = e;
            }else if(e < min)
            {
                min = e;
            }
        }

        if(min == max)
        {
            return 1;
        }

        for(int i = 0; i < N; i++)
        {
            if(arr[i] == min)
            {
                minIdx = i;
            }else if(arr[i] == max)
            {
                maxIdx = i;
            }

            if(minIdx != -1 && maxIdx != -1)
            {
                len = Math.min(len, Math.abs(minIdx - maxIdx) + 1);
            }
        }
        
        
        return len;
    }
}
