public class Solution 
{
    public int solve(String str)
    {
        int count = 0;
        int N = str.length();
        final int MOD =  10003;

        for(int i = 0; i < N; i++)
        {
            if(isVoewl(str.charAt(i)))
            {
                count = (count + (N - i)) % MOD;
            }
        }

        return count % MOD;
    }

    public boolean isVoewl(char ch)
    {
        if(ch == 'a' || ch == 'e' ||ch == 'i' ||ch == 'o' ||ch == 'u' ||
            ch == 'A' ||ch == 'E' ||ch == 'I' ||ch == 'O' ||ch == 'U' 
        )
        {
            return true;
        }
        
        return false;
    }
}
