public class Solution 
{
    public int solve(String str)
    {
        int N = str.length();
        int count = 0;
        int ans = 0;
        final int MOD = 1000000007; 
        
        for(int i = N - 1; i >= 0; i--)
        {
            if(str.charAt(i) == 'G')
            {
                count++;
            }else if(str.charAt(i) == 'A')
            {
                ans = (ans + count) % MOD;
            }
        }

        return ans % MOD;
    }
}
