=====================Solution 1=====================
Time : O(N^2)   // TLE Problem with this solution
Space : O(1)
public class Solution
{
    // DO NOT MODIFY THE ARGUMENTS WITH "final" PREFIX. IT IS READ ONLY
    public int maxSubArray(final int[] arr)
    {
        int maxSum = Integer.MIN_VALUE;
        int sum = 0;
        int N = arr.length;

        for(int i = 0; i < N; i++)
        {
            sum = 0;
            for(int j = i; j < N; j++)
            {
                if(arr[i] < 0 && (sum - arr[i] == 0))
                {
                    break;
                }

                sum = sum + arr[j];

                if(maxSum < sum)
                {
                    maxSum = sum;
                }
            }
        }

        return maxSum;
    }
}
=====================Solution 2=====================
Time : O(N)
Space : O(1)
public class Solution
{
    // DO NOT MODIFY THE ARGUMENTS WITH "final" PREFIX. IT IS READ ONLY
    public int maxSubArray(final int[] arr)
    {
        int maxSum = Integer.MIN_VALUE;
        int sum = 0;
        int N = arr.length;

        for(int i = 0; i < N; i++)
        {
            sum = sum + arr[i];
			maxSum = Math.max(sum, maxSum);

            if(sum < 0)
			{
				sum = 0;
			}
        }

        return maxSum;
    }
}
