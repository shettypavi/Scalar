public class Solution 
{
    public int maxSubarray(int N, int B, int[] arr)
    {
        int sum = 0;
        int maxSum = 0;

        for(int i = 0; i < N; i++)
        {
            sum = 0;
            for(int j = i; j < N; j++)
            {
                sum = sum + arr[j];
                
                if(sum <= B)
                {
                    maxSum = Math.max(sum, maxSum);
                }else
                {
                    break;
                }
            }
        }

        return maxSum;
    }
}
