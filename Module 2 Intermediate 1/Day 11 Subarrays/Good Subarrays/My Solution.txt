public class Solution 
{
    public int solve(int[] arr, int B) 
    {
        int count = 0;
        int sum = 0;
        int N = arr.length;

        for(int i = 0; i < N; i++)
        {
            sum = 0;
            for(int j = i; j < N; j++)
            {
                sum = sum + arr[j];

                if((j - i + 1) % 2 == 0)
                {
                    if(sum < B)
                    {
                        count++;
                    }
                }else
                {
                    if(sum > B)
                    {
                        count++;
                    }
                }
            }
        }

        return count;
    }
}
