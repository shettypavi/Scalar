public class Solution {
    public int solve(int[] A, int B) {
        int n  = A.length;
        int dp[] = new int[n+1];
        dp[0]=0;
        int ans=0;
        for(int i=0;i<n;i++)dp[i+1]=dp[i]+A[i];
        for(int i=0;i<n;i++){
            for (int j=i;j<n;j++){
                int sz=j-i+1;
                int sum=dp[j+1]-dp[i];
                if(sz%2==0 && sum<B)ans++;
                if(sz%2==1 && sum>B)ans++;
            }
        }
        return ans;
    }
}
