✅ Final Answer
- Worst case (when B ≈ N/2): O(N²)
- Space complexity: O(N)

public class Solution 
{
    public ArrayList<Integer> solve(ArrayList<Integer> arr, int B) 
    {
        ArrayList<Integer> ans = new ArrayList<Integer>();
        int L = (2 * B) + 1;
        boolean flag;
        int N = arr.size();
        int state;

        for(int i = 0; i <= N - L; i++)
        {
            flag = true;
            state = arr.get(i);

            for(int j = i; j < i + L; j++)
            {
                if(arr.get(j) == state)
                {
                    state = 1 - state;   
                }else
                {
                    flag = false;
                    break;
                }
            }

            if(flag == true)
            {
                ans.add(i + B);
            }
        }

        return ans;
    }
}
