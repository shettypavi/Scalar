===================Java Solution===================
public class Solution {
    public int compressBits(int[] A) {
        int ans = 0;
        for(int i = 0;i < A.length; i++){
            ans ^= A[i];
        }
        return ans;
    }
}
===================C++ Solution===================

===================C# Solution===================

    public int compressBits(List<int> A) {
        int ans = 0;

            ans ^= A[i];

    }
}
