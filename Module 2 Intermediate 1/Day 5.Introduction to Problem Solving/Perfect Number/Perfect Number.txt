===================Solution 1===================
Time O(A*N) where N = Total number of input and A = Max size if give input
Space O(1)

import java.lang.*;
import java.util.*;

public class Main 
{
    public static void main(String[] args) 
    {
        // YOUR CODE GOES HERE
        // Please take input and print output to standard input/output (stdin/stdout)
        // DO NOT USE ARGUMENTS FOR INPUTS
        // E.g. 'Scanner' for input & 'System.out' for output
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();

        for(int i = 1; i <= A; i++)
        {
            int N = sc.nextInt();
            int sum = 0;

            for(int j = 1; j < N; j++)
            {
                if(N % j == 0)
                {
                    sum = sum + j;
                }
            }

            if(sum == N)
            {
                System.out.println("YES");
            }else
            {
                System.out.println("NO");
            }
        }
    }
}
===================Solution 2===================
Time O(A*Sqrt(N)) where N = Total number of input and A = Max size if give input
Space O(1)
import java.lang.*;
import java.util.*;

public class Main 
{
    public static void main(String[] args) 
    {
        // YOUR CODE GOES HERE
        // Please take input and print output to standard input/output (stdin/stdout)
        // DO NOT USE ARGUMENTS FOR INPUTS
        // E.g. 'Scanner' for input & 'System.out' for output
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();

        for(int i = 1; i <= N; i++)
        {
            int ip = sc.nextInt();
            int sum = 1;

            for(int j = 2; j <= (int)Math.sqrt(ip); j++)
            {
                if(ip % j == 0)
                {
                    if(j == (ip / j))
                    {
                        sum = sum + j;
                    }else
                    {
                        sum = sum + j + (ip / j);
                    }
                }
            }

            if(sum == ip)
            {
                System.out.println("YES");
            }else
            {
                System.out.println("NO");
            }
        }
    }
}
