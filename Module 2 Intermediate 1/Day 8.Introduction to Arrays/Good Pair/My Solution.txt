/*
Time : O(N^2)
*/
public class Solution 
{
    public int solve(int[] arr, int B) 
    {
        if(isGoodPair(arr, B))
        {
            return 1;
        }

        return 0;
    }

    public boolean isGoodPair(int[] arr, int B)
    {
        int N = arr.length;

        for(int i = 0; i < N; i++)
        {
            for(int j = i + 1; j < N; j++)
            {
                if(arr[i] + arr[j] == B)
                {
                    return true;
                }
            }
        }

        return false;
    }
}
