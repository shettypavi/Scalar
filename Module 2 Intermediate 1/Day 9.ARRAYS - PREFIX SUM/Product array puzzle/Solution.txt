public class Solution {
    public int[] solve(int[] A) {
        int product = 1;
        int n = A.length;
        for(int i = 0; i < A.length; i++)
        {
            product *= A[i];
        }
        
        int []ans = new int[n];
        for(int i = 0; i < A.length; i++) ans[i] = product/A[i];
        return ans;
    }
}
