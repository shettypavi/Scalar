/*
Time : O(2N) = O(N)
Space : O(1)
*/
public class Solution 
{
    public int solve(int[] arr, int B) 
    {
        int N = arr.length;
        long sum = 0;
        long maxSum = 0;

        for(int i = 0; i < B; i++)
        {
            sum = sum + arr[i];
        }

        maxSum = sum;
        B = B - 1;

        for(int i = N - 1; B >= 0 && i > 0; i--, B--)
        {
            sum = sum - arr[B] + arr[i];

            if(sum > maxSum)
            {
                maxSum = sum;
            }
        }

        return (int)maxSum;
    }
}
