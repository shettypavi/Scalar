============================Java Solution============================
Method 1:

public class Solution {
    public int numSetBits(long A) {
        int total_ones = 0;
        while (A != 0) {
            // rightmost set bit becomes unset
            A = A & (A - 1);
            total_ones++;
        }
        return total_ones;
    }
}

Method 2:

public class Solution {
    public int numSetBits(long A) {
        int total_ones = 0;
        for(int i = 0; i < 30; i++){
            if((A & (1 << i)) > 0){
                total_ones += 1;
            }
        }
        return total_ones;
    }
}
