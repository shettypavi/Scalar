public class Solution {
	public String addBinary(String A, String B) {
	    int nA, nB;
	    char res[];
	    int i, j, k;
	    nA = A.length();
	    nB = B.length();
	    res = new char[Math.max(nA, nB) + 1];
	    k = Math.max(nA, nB);
	    i = nA - 1;
	    j = nB - 1;
	    int sum = 0, carry = 0;
	    while (i >= 0 || j >= 0 || carry != 0) {
	        sum = carry;
	        if (i >= 0)
	            sum += (A.charAt(i) - '0');
	        if (j >= 0)
	            sum += (B.charAt(j) - '0');
	        res[k] = (char) ((sum % 2) + '0');
	        carry = sum / 2;
	        i--;
	        j--;
	        k--;
	    }
	    if (res[0] == '1')
	        return new String(res);
	    int len = Math.max(nA, nB);
	    return new String(res, 1, len);
	}
}

using System.Text;
class Solution {
    public string addBinary(string A, string B) {
        List < char > l1 = new List < char > ();
        int ansLen = 0, carry = 0, sum = 0;
        int i = A.Length - 1;
        int j = B.Length - 1;
        // we add bits from the rightmost bit to the leftmost bit
        while (i >= 0 || j >= 0 || carry > 0) {
            sum = carry;
            if (i >= 0)
                sum += (A[i] - '0');
            if (j >= 0)
                sum += (B[j] - '0');
            l1.Add((char)('0' + sum % 2));
            carry = sum / 2;
            i--;
            j--;
        }
        StringBuilder ans = new StringBuilder();
        for (i = l1.Count - 1; i >= 0; i--) {
            ans.Append(l1[i]);
        }
        return ans.ToString();
    }
}

def get(x):
    if x == 0:
        return "0"
    if x == 1:
        return "1"
        
class Solution:
    # @param A : string
    # @param B : string
    # @return a strings
    def addBinary(self, A, B):
        n = -len(A)
        m = -len(B)
        nm = min(n, m)
        carry = 0
        ans = ""
        # we add bits from the rightmost bit to the leftmost bit
        for i in range(-1, nm - 1, -1):
            if i >= n and i >= m:
                v = ord(A[i]) + ord(B[i]) - 2 * ord('0') + carry
                carry = v // 2
                v -= 2 * carry
                ans += get(v)
            elif i >= n:
                v = ord(A[i]) - ord('0') + carry
                carry = v // 2
                v -= 2 * carry
                ans += get(v)
            else:
                v = ord(B[i]) - ord('0') + carry
                carry = v // 2
                v -= 2 * carry
                ans += get(v)
        if carry == 1:
            ans += get(carry)
        return ans[::-1]

/**
 * @input A : String termination by '\0'
 * @input B : String termination by '\0'
 * 
 * @Output string. Make sure the string ends with null character
 */
char* addBinary(char* A, char* B) {
    int len = (strlen(A) >= strlen(B)) ? strlen(A) : strlen(B);
    char *C = (char *)malloc((len + 5) * sizeof(char));
    int i = strlen(A) - 1, j = strlen(B) - 1, k = 0, carry = 0, sum;
    // we add bits from the rightmost bit to the leftmost bit
    while (i >= 0 || j >= 0 || carry) {
        sum = carry;
        if (i >= 0) 
            sum += (A[i] - '0');
        if (j >= 0) 
            sum += (B[j] - '0');
        C[k] = (char)('0' + sum % 2);
        carry = sum / 2;
        i--; 
        j--;
        k++;
    }
    C[k] = '\0';
    for(i = 0, j = k - 1 ; i <= j ; i++, j--){
        char ch = C[i];
        C[i] = C[j];
        C[j] = ch;
    }
    return C;
}
